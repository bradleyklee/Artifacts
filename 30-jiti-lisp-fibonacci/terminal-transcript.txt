FIBONACCI (WORLD-G182::N)
Return the nth Fibonacci number exactly using fast doubling. N must be a 
nonnegative integer.
(DEFUN FIBONACCI (N)
  "Return the nth Fibonacci number exactly using fast doubling. N must be a 
nonnegative integer."
  (CHECK-TYPE N (INTEGER 0 *))
  (LET ((A 0) (B 1))
    (LOOP FOR BIT DOWNFROM (1- (INTEGER-LENGTH N)) TO 0
          DO (LET* ((C (* A (- (* 2 B) A))) (D (+ (* A A) (* B B))))
               (IF (LOGBITP BIT N)
                   (SETF A D
                         B (+ C D))
                   (SETF A C
                         B D))))
    A))
